async function hashPw(salt, pw) {
  return hex(await crypto.subtle.digest('SHA-256', enc.encode('soda-sprechen|' + salt + '|' + norm(pw))));
}
function same(a, b) {
  if (a.length !== b.length) return false;
  let d = 0; for (let i = 0; i < a.length; i++) d |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return d === 0;
}
function randHex(n) { const u = new Uint8Array(n); crypto.getRandomValues(u); return hex(u); }

function corsHeaders(req, env) {
  const origin = req.headers.get('Origin') || '';
  const list = (env.ORIGINS || '').split(',').map(s => s.trim()).filter(Boolean);
  let allow = '*';
  if (list.length) allow = list.includes(origin) ? origin : list[0];
  return {
    'Access-Control-Allow-Origin': allow,
    'Access-Control-Allow-Methods': 'POST, GET, OPTIONS',
    'Access-Control-Allow-Headers': 'Content-Type',
    'Access-Control-Max-Age': '86400',
    'Vary': 'Origin'
  };
}
function json(req, env, status, obj) {
  return new Response(JSON.stringify(obj), { status, headers: { 'Content-Type': 'application/json; charset=utf-8', 'Cache-Control': 'no-store', ...corsHeaders(req, env) } });
}

async function checkLock(env, keys) {
  const t = now();
  for (const k of keys) {
    const r = await env.DB.prepare('SELECT n, until FROM fails WHERE k = ?1').bind(k).first();
    if (r && r.until > t) throw new HttpError(429, 'locked', '로그인에 여러 번 실패했습니다. ' + Math.ceil((r.until - t) / 1000) + '초 뒤에 다시 시도해 주십시오.', { retry: Math.ceil((r.until - t) / 1000) });
  }
}
async function addFail(env, k, limit, lockMs) {
  const t = now();
  const r = await env.DB.prepare('SELECT n FROM fails WHERE k = ?1').bind(k).first();
  const n = (r ? r.n : 0) + 1;
  if (n >= limit) await env.DB.prepare('INSERT INTO fails(k, n, until) VALUES(?1, 0, ?2) ON CONFLICT(k) DO UPDATE SET n = 0, until = ?2').bind(k, t + lockMs).run();
  else await env.DB.prepare('INSERT INTO fails(k, n, until) VALUES(?1, ?2, 0) ON CONFLICT(k) DO UPDATE SET n = ?2').bind(k, n).run();
}

const BAD_LOGIN = '아이디 또는 비밀번호가 맞지 않습니다.';
async function authenticate(env, req, body) {
  const id = idKey(body.id), pw = norm(body.pw);
  if (!id || !pw || id.length > 40 || pw.length > 80) throw new HttpError(401, 'bad_login', BAD_LOGIN);
  const ip = req.headers.get('CF-Connecting-IP') || 'unknown';
  const uk = 'u:' + id, ik = 'i:' + ip;
  await checkLock(env, [uk, ik]);
  const acc = await env.DB.prepare('SELECT id, name, salt, hash, levels, max, admin FROM accounts WHERE id = ?1').bind(id).first();
  const h = await hashPw(acc ? acc.salt : '0'.repeat(32), pw);
  if (!acc || !same(h, acc.hash)) {
    await addFail(env, uk, 5, LOCK_MS);
    await addFail(env, ik, 30, IP_LOCK_MS);
    throw new HttpError(401, 'bad_login', BAD_LOGIN);
  }
  await env.DB.prepare('DELETE FROM fails WHERE k = ?1').bind(uk).run();
  return { id: acc.id, name: acc.name, levels: JSON.parse(acc.levels), max: acc.max, admin: !!acc.admin };
}

function checkExam(acc, exam, teil) {
  const ex = EXAM_MAP[exam];
  if (!ex) throw new HttpError(400, 'bad_exam', '알 수 없는 시험입니다.');
  const t = Number(teil);
  if (!Number.isInteger(t) || t < 1 || t > ex.teile) throw new HttpError(400, 'bad_teil', '알 수 없는 Teil입니다.');
  if (!acc.levels.includes(ex.access)) throw new HttpError(403, 'no_access', '이 시험은 열려 있지 않습니다.');
  return t;
}

async function usageOf(env, id) {
  const rs = await env.DB.prepare('SELECT exam, teil, COUNT(*) AS c FROM attempts WHERE id = ?1 GROUP BY exam, teil').bind(id).all();
  const used = {};
  for (const r of rs.results || []) used[r.exam + '|' + r.teil] = r.c;
  return used;
}

async function hStart(env, req, body) {
  const acc = await authenticate(env, req, body);
  const exam = String(body.exam || ''), teil = checkExam(acc, exam, body.teil);
  let reserved = false;
  for (let i = 0; i < 4 && !reserved; i++) {
    try {
      const r = await env.DB.prepare(
        'INSERT INTO attempts(id, exam, teil, no, at) SELECT ?1, ?2, ?3, COALESCE(MAX(no), 0) + 1, ?4 FROM attempts WHERE id = ?1 AND exam = ?2 AND teil = ?3 HAVING COALESCE(MAX(no), 0) < ?5'
      ).bind(acc.id, exam, teil, now(), acc.max).run();
      if (!r.meta.changes) throw new HttpError(409, 'used_up', '이 Teil의 연습 횟수를 모두 사용했습니다.');
      reserved = true;
    } catch (e) {
      if (e instanceof HttpError) throw e;
      if (i === 3) throw e;                       // 동시에 시작한 요청과 부딪힌 경우에만 다시 시도합니다.
    }
  }
  const row = await env.DB.prepare('SELECT MAX(no) AS no, COUNT(*) AS c FROM attempts WHERE id = ?1 AND exam = ?2 AND teil = ?3').bind(acc.id, exam, teil).first();
  const ticket = randHex(24);
  await env.DB.prepare('DELETE FROM tickets WHERE exp < ?1').bind(now()).run();
  await env.DB.prepare('INSERT INTO tickets(code, id, exam, teil, no, exp, ex_left, pa_left) VALUES(?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8)')
    .bind(ticket, acc.id, exam, teil, row.no, now() + TICKET_TTL_MS, TICKET_EXAMINER, TICKET_PARTNER).run();
  return { ok: true, no: row.no, used: row.c, ticket };
}

async function signedUrl(env, agentId) {
  if (!agentId || !env.ELEVENLABS_API_KEY) throw new HttpError(500, 'config', '서버 설정이 끝나지 않았습니다. 선생님께 알려 주십시오.');
  let r;
  try {
    r = await fetch('https://api.elevenlabs.io/v1/convai/conversation/get-signed-url?agent_id=' + encodeURIComponent(agentId), { headers: { 'xi-api-key': env.ELEVENLABS_API_KEY } });
  } catch (e) { throw new HttpError(502, 'upstream', '음성 서버에 연결하지 못했습니다. 잠시 뒤에 다시 시도해 주십시오.'); }
  if (!r.ok) throw new HttpError(502, 'upstream', '음성 서버가 연결을 허가하지 않았습니다(' + r.status + '). 선생님께 알려 주십시오.');
  const j = await r.json().catch(() => null);
  if (!j || typeof j.signed_url !== 'string' || !j.signed_url.startsWith('wss://api.elevenlabs.io/')) throw new HttpError(502, 'upstream', '음성 서버의 응답이 올바르지 않습니다.');
  return j.signed_url;
}

async function hSign(env, req, body) {
  const acc = await authenticate(env, req, body);
  const role = String(body.role || '');
  if (role === 'examiner' || role === 'partner') {
    const col = role === 'examiner' ? 'ex_left' : 'pa_left';
    const r = await env.DB.prepare('UPDATE tickets SET ' + col + ' = ' + col + ' - 1 WHERE code = ?1 AND id = ?2 AND exp > ?3 AND ' + col + ' > 0')
      .bind(String(body.ticket || ''), acc.id, now()).run();
    if (!r.meta.changes) throw new HttpError(403, 'no_ticket', '연습 시작 허가가 없거나 만료되었습니다. 처음부터 다시 시작해 주십시오.');
    try {
      return { url: await signedUrl(env, role === 'examiner' ? env.AGENT_EXAMINER : env.AGENT_PARTNER) };
    } catch (e) {
      await env.DB.prepare('UPDATE tickets SET ' + col + ' = ' + col + ' + 1 WHERE code = ?1').bind(String(body.ticket)).run();
      throw e;
    }
  }
  if (role === 'evaluator') {
    const exam = String(body.exam || ''), teil = checkExam(acc, exam, body.teil);
    const has = await env.DB.prepare('SELECT 1 AS x FROM attempts WHERE id = ?1 AND exam = ?2 AND teil = ?3 LIMIT 1').bind(acc.id, exam, teil).first();
    if (!has) throw new HttpError(403, 'no_attempt', '연습 기록이 없는 Teil은 평가할 수 없습니다.');
    const cap = Math.max(4, acc.max * 2);
    const r = await env.DB.prepare('INSERT INTO evals(id, exam, teil, n) VALUES(?1, ?2, ?3, 1) ON CONFLICT(id, exam, teil) DO UPDATE SET n = n + 1 WHERE n < ?4')
      .bind(acc.id, exam, teil, cap).run();
    if (!r.meta.changes) throw new HttpError(429, 'eval_limit', '이 Teil의 평가 요청 횟수를 모두 사용했습니다.');
    try {
      return { url: await signedUrl(env, env.AGENT_EVALUATOR) };
    } catch (e) {
      await env.DB.prepare('UPDATE evals SET n = n - 1 WHERE id = ?1 AND exam = ?2 AND teil = ?3 AND n > 0').bind(acc.id, exam, teil).run();
      throw e;
    }
  }
  throw new HttpError(400, 'bad_role', '알 수 없는 요청입니다.');
}

async function hRefund(env, req, body) {
  const acc = await authenticate(env, req, body);
  const code = String(body.ticket || '');
  const tk = await env.DB.prepare('SELECT exam, teil, no FROM tickets WHERE code = ?1 AND id = ?2').bind(code, acc.id).first();
  if (!tk) throw new HttpError(403, 'no_ticket', '돌려줄 시도를 찾지 못했습니다.');
  const cap = acc.admin ? 9999 : REFUND_CAP;
  const r = await env.DB.prepare('INSERT INTO refunds(id, exam, teil, n) VALUES(?1, ?2, ?3, 1) ON CONFLICT(id, exam, teil) DO UPDATE SET n = n + 1 WHERE n < ?4')
    .bind(acc.id, tk.exam, tk.teil, cap).run();
  if (!r.meta.changes) throw new HttpError(429, 'refund_limit', '이 Teil은 더 이상 횟수를 돌려받을 수 없습니다.');
  await env.DB.prepare('DELETE FROM tickets WHERE code = ?1').bind(code).run();
  await env.DB.prepare('DELETE FROM attempts WHERE id = ?1 AND exam = ?2 AND teil = ?3 AND no = ?4').bind(acc.id, tk.exam, tk.teil, tk.no).run();
  const c = await env.DB.prepare('SELECT COUNT(*) AS c FROM attempts WHERE id = ?1 AND exam = ?2 AND teil = ?3').bind(acc.id, tk.exam, tk.teil).first();
  return { ok: true, used: c.c };
}

async function hAdminUsage(env, req, body) {
  const acc = await authenticate(env, req, body);
  if (!acc.admin) throw new HttpError(403, 'no_access', '관리자만 볼 수 있습니다.');
  const accs = (await env.DB.prepare('SELECT id, name, levels, max, admin FROM accounts ORDER BY admin DESC, name').all()).results || [];
  const rs = (await env.DB.prepare('SELECT id, exam, teil, COUNT(*) AS c, MAX(at) AS last FROM attempts GROUP BY id, exam, teil').all()).results || [];
  const ev = (await env.DB.prepare('SELECT id, exam, teil, n FROM evals').all()).results || [];
  const byId = {};
  for (const a of accs) byId[a.id] = { name: a.name, levels: JSON.parse(a.levels), max: a.max, admin: !!a.admin, usage: {}, evals: {} };
  for (const r of rs) if (byId[r.id]) byId[r.id].usage[r.exam + '|' + r.teil] = { n: r.c, last: r.last };
  for (const r of ev) if (byId[r.id]) byId[r.id].evals[r.exam + '|' + r.teil] = r.n;
  return { accounts: accs.map(a => byId[a.id]) };
}

async function hAdminReset(env, req, body) {
  const acc = await authenticate(env, req, body);
  if (!acc.admin) throw new HttpError(403, 'no_access', '관리자만 할 수 있습니다.');
  const target = idKey(body.target);
  const t = await env.DB.prepare('SELECT id FROM accounts WHERE id = ?1').bind(target).first();
  if (!t) throw new HttpError(404, 'no_account', '그 아이디가 없습니다.');
  const where = ['id = ?1'], args = [target];
  if (body.exam) {
    if (!EXAM_MAP[body.exam]) throw new HttpError(400, 'bad_exam', '알 수 없는 시험입니다.');
    where.push('exam = ?2'); args.push(String(body.exam));
    if (body.teil) { where.push('teil = ?3'); args.push(Number(body.teil)); }
  }
  const w = where.join(' AND ');
  const a = await env.DB.prepare('DELETE FROM attempts WHERE ' + w).bind(...args).run();
  await env.DB.prepare('DELETE FROM evals WHERE ' + w).bind(...args).run();
  await env.DB.prepare('DELETE FROM tickets WHERE ' + w).bind(...args).run();
  await env.DB.prepare('DELETE FROM refunds WHERE ' + w).bind(...args).run();
  return { ok: true, removed: a.meta.changes };
}

    list('다음 연습', ev.next, 8);
    if (ev.comparison) { h += '<h3 style="margin:16px 0 4px">시도 비교</h3><p style="margin:0">' + eh(ev.comparison, 800) + '</p>'; tx += '\n[시도 비교]\n' + clean(ev.comparison, 800) + '\n'; }
  }
  let budget = 14000;
  for (const a of arr(sm.attempts, 2)) {
    if (!a || typeof a !== 'object') continue;
    h += '<h3 style="margin:20px 0 4px;border-top:2px solid #222;padding-top:10px">' + eh(num(a.no), 4) + '번째 시도</h3>';
    h += '<p style="margin:0 0 6px;color:#444">' + [
      '대화 시간 ' + mmss(a.durationSec), '학생 단어 ' + num(a.words), a.share !== '' && a.share != null ? '발화 비율 ' + num(a.share) + '%' : '', '가장 긴 발화 ' + num(a.longest) + '단어',
      '고정 표현 ' + num(a.phrases) + '종', '머뭇거림 ' + num(a.fillers) + '회'].filter(Boolean).map(x => hesc(x)).join(' · ') + '</p>';
    h += '<p style="margin:0 0 6px;color:#444">' + eh(a.reason, 200) + (a.partner ? ' · 파트너 ' + eh(a.partner, 30) : '') + '</p>';
    if (a.topic) h += '<p style="margin:0 0 6px;color:#444">고른 주제·카드: ' + eh(a.topic, 400) + '</p>';
    tx += '\n=== ' + num(a.no) + '번째 시도 · ' + mmss(a.durationSec) + ' · 학생 ' + num(a.words) + '단어 ===\n';
    const ck = arr(a.checks, 14);
    if (ck.length) {
      h += '<div style="margin:6px 0">' + ck.map(c => '<div>' + (MARK[c.status] || '·') + ' ' + eh(c.ko, 200) + (c.note ? ' <span style="color:#777;font-size:13px">— ' + eh(c.note, 200) + '</span>' : '') + '</div>').join('') + '</div>';
      tx += ck.map(c => (MARK[c.status] || '·') + ' ' + clean(c.ko, 200)).join('\n') + '\n';
    }
    let t = '';
    for (const b of arr(a.transcript, 120)) {
      if (!b || budget <= 0) break;
      const line = clean(b.text, 1200).slice(0, budget); budget -= line.length;
      h += (t === '' ? '<div style="margin-top:8px;padding:8px 12px;background:#f5f5f5;border-radius:6px">' : '') + '<div style="margin:0 0 6px"><b>' + eh(b.who, 12) + '</b> ' + hesc(line) + '</div>';
      t += clean(b.who, 12) + ': ' + line + '\n';
    }
    if (t) { h += '</div>'; tx += '\n' + t; }
    if (budget <= 0) h += '<p style="color:#777;font-size:13px">전사가 길어 일부만 표시했습니다. 전체는 첨부 파일에 있습니다.</p>';
  }
  h += '<p style="margin-top:20px;color:#777;font-size:13px">점수와 등급은 전사를 바탕으로 한 SODA의 추정이며 공식 채점이 아닙니다. 전체 내용은 첨부 파일(HTML)을 열어 확인해 주십시오.</p></div>';
  return { html: h, text: tx };
}

async function hReport(env, req, body) {
  const acc = await authenticate(env, req, body);
  const exam = String(body.exam || ''), teil = checkExam(acc, exam, body.teil);
  const kind = body.kind === 'eval' ? 'eval' : 'attempt';
  const has = await env.DB.prepare('SELECT 1 AS x FROM attempts WHERE id = ?1 AND exam = ?2 AND teil = ?3 LIMIT 1').bind(acc.id, exam, teil).first();
  if (!has) throw new HttpError(403, 'no_attempt', '연습 기록이 없는 Teil의 결과는 보낼 수 없습니다.');
  if (!env.MAIL_URL || !env.MAIL_KEY) throw new HttpError(503, 'mail_off', '결과 전송 설정이 끝나지 않았습니다. 선생님께 알려 주십시오.');
  const file = body.file && typeof body.file === 'object' ? body.file : null;
  if (file && String(file.html || '').length > FILE_MAX) throw new HttpError(413, 'too_large', '결과 파일이 너무 큽니다.');
  const day = kst(now()).slice(0, 10), cap = acc.admin ? MAIL_DAILY_ADMIN : MAIL_DAILY;
  const q = await env.DB.prepare('INSERT INTO mails(id, day, n) VALUES(?1, ?2, 1) ON CONFLICT(id, day) DO UPDATE SET n = n + 1 WHERE n < ?3').bind(acc.id, day, cap).run();
  if (!q.meta.changes) throw new HttpError(429, 'mail_limit', '오늘 보낼 수 있는 결과 전송 횟수를 모두 사용했습니다. 내일 다시 시도해 주십시오.');
  const m = mailBodies(acc, { kind, exam_label: body.exam_label, teil_label: body.teil_label, summary: body.summary });
  const no = kind === 'attempt' && body.summary && Array.isArray(body.summary.attempts) && body.summary.attempts[0] ? num(body.summary.attempts[0].no) : '';
  const subject = '[SODA 말하기] ' + clean(acc.name, 40) + ' · ' + clean(body.exam_label, 60) + ' ' + clean(body.teil_label, 60).split(' · ')[0] + ' · ' + (kind === 'eval' ? '평가 결과' : (no ? no + '번째 ' : '') + '연습 기록');
  const payload = { key: env.MAIL_KEY, subject: clean(subject, 200), text: m.text, html: m.html };
  if (file && file.html) payload.file = { name: clean(file.name || 'SODA 말하기 결과.html', 120).replace(/[\\/:*?"<>|_]/g, ' ').replace(/\s+/g, ' '), html: String(file.html) };
  const undo = () => env.DB.prepare('UPDATE mails SET n = n - 1 WHERE id = ?1 AND day = ?2 AND n > 0').bind(acc.id, day).run().catch(() => { });
  let r, j = null;
  try {
    r = await fetch(env.MAIL_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify(payload), redirect: 'follow', signal: AbortSignal.timeout(25000) });
    j = await r.json().catch(() => null);
  } catch (e) { await undo(); throw new HttpError(502, 'mail_fail', '메일 서버에 연결하지 못했습니다. 잠시 뒤에 다시 시도해 주십시오.'); }
  if (!r.ok || !j || j.ok !== true) { await undo(); console.log('mail relay', r.status, j && j.error); throw new HttpError(502, 'mail_fail', '메일 서버가 발송을 거절했습니다. 잠시 뒤에 다시 시도해 주십시오.'); }
  return { ok: true };
}

const ROUTES = { '/login': hLogin, '/start': hStart, '/sign': hSign, '/refund': hRefund, '/report': hReport, '/admin/usage': hAdminUsage, '/admin/reset': hAdminReset };
const BODY_MAX = { '/report': 420 * 1000 };

export default {
  async fetch(req, env) {
    const url = new URL(req.url);
    if (req.method === 'OPTIONS') return new Response(null, { status: 204, headers: corsHeaders(req, env) });
    if (req.method === 'GET' && (url.pathname === '/' || url.pathname === '/health')) return json(req, env, 200, { ok: true, service: 'soda-sprechen' });
    const h = ROUTES[url.pathname];
    if (!h || req.method !== 'POST') return json(req, env, 404, { error: 'not_found', message: '없는 주소입니다.' });
    try {
      const text = await req.text();
      if (text.length > (BODY_MAX[url.pathname] || 4000)) throw new HttpError(413, 'too_large', '요청이 너무 큽니다.');
      let body; try { body = JSON.parse(text || '{}'); } catch (e) { throw new HttpError(400, 'bad_json', '요청 형식이 올바르지 않습니다.'); }
      if (!body || typeof body !== 'object') throw new HttpError(400, 'bad_json', '요청 형식이 올바르지 않습니다.');
      return json(req, env, 200, await h(env, req, body));
    } catch (e) {
      if (e instanceof HttpError) return json(req, env, e.status, { error: e.code, message: e.message, ...e.extra });
      console.log('error', e && e.stack || e);
      return json(req, env, 500, { error: 'server', message: '서버에 문제가 생겼습니다. 잠시 뒤에 다시 시도해 주십시오.' });
    }
  }
};
